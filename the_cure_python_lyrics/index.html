# Olivia Rodrigo - The Cure (Lyrics Animator)
import sys, time, os

def typewriter(text, speed=0.075, color="\033[95m"):
    sys.stdout.write(color)
    for char in text:
        sys.stdout.write(char)
        sys.stdout.flush()
        time.sleep(speed)
    print("\033[0m")

lyrics = [
    ("And my head is full of poison", 0.075, 0.5),
    ("And my heart is full of doubt", 0.080, 0.6),
    ("I got toxins in my bloodstream", 0.070, 0.4),
    ("You tried hard to suck 'em out", 0.075, 0.8),
    ("And it feels like medication", 0.080, 0.5),
    ("And it's good for me, I'm sure", 0.080, 0.7),
    ("But it don't matter how your love feels anymore", 0.060, 0.6),
    ("It'll never be the cure", 0.095, 0.8),
    ("It'll never be the cure...", 0.110, 1.5)
]

for line, speed, pause in lyrics:
    typewriter(line, speed)
    time.sleep(pause)
